### STEP 10: Test Backend API Directly
--------------------------------------
Open PowerShell and test the API:

# Get your JWT token from browser console:
# localStorage.getItem('supabase.auth.token')

$token = "YOUR_JWT_TOKEN_HERE"
$headers = @{
    "Content-Type" = "application/json"
    "Authorization" = "Bearer $token"
}

$body = @{
    phone_number = "+252612345678"
    company_name = "Test Company Ltd"
    company_registration_number = "REG-TEST-001"
    company_email = "test@company.com"
    company_phone = "+252-1-111111"
    region = "mogadishu"
    landmark = "Business District"
} | ConvertTo-Json

Invoke-RestMethod -Uri "http://localhost:3001/api/addresses/generate-company" -Method Post -Headers $headers -Body $body

Expected Result:
✓ Status: 201 Created
✓ Response includes company address with all details
✓ address_type = "company"
✓ kivro_code starts with "KV-BIZ-"

-- Check address types
SELECT 
    address_type,
    COUNT(*) as count
FROM kivro_addresses
GROUP BY address_type;

personal | 1
company  | 1

    kivro_code,
    display_address,
    company_name,
    company_registration_number,
    company_email,
    is_active
